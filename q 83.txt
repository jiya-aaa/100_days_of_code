Q83. Count vowels and consonants in a string.

#include <stdio.h>
int main(void) {
    char s[200];int v=0,c=0;fgets(s,sizeof(s),stdin);for(int i=0;s[i];i++){char x=s[i];if((x>='a'&&x<='z')||(x>='A'&&x<='Z')){if(x=='a'||x=='e'||x=='i'||x=='o'||x=='u'||x=='A'||x=='E'||x=='I'||x=='O'||x=='U')v++;else c++;}}printf("Vowels = %d\nConsonants = %d\n",v,c);
    return 0;
}
