#include <stdio.h>

enum Status {
    SUCCESS,
    FAILURE,
    TIMEOUT
};

int main(void) {
    enum Status s = SUCCESS;

    switch (s) {
        case SUCCESS: printf("Operation completed successfully.\n"); break;
        case FAILURE: printf("Operation failed.\n"); break;
        case TIMEOUT: printf("Operation timed out.\n"); break;
    }
    return 0;
}
