Q69. Find the second largest element in an array.

#include <stdio.h>
#include <limits.h>
int main(void){int n,a[100],max=INT_MIN,second=INT_MIN;scanf("%d",&n);for(int i=0;i<n;i++){scanf("%d",&a[i]);if(a[i]>max){second=max;max=a[i];}else if(a[i]>second&&a[i]!=max)second=a[i];}if(second==INT_MIN)puts("No second largest element");else printf("Second largest = %d\n",second);return 0;}
