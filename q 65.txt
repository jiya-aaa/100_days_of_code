Q65. Search in a sorted array using binary search.

#include <stdio.h>
int main(void){int n,a[100],x,l=0,r,mid,pos=-1;scanf("%d",&n);for(int i=0;i<n;i++)scanf("%d",&a[i]);scanf("%d",&x);r=n-1;while(l<=r){mid=(l+r)/2;if(a[mid]==x){pos=mid;break;}if(a[mid]<x)l=mid+1;else r=mid-1;}printf("%d\n",pos);return 0;}
