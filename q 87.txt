Q87. Count spaces, digits, and special characters in a string.

#include <stdio.h>
int main(void) {
    char s[200];int sp=0,d=0,sc=0;fgets(s,sizeof(s),stdin);for(int i=0;s[i];i++){if(s[i]==' ')sp++;else if(s[i]>='0'&&s[i]<='9')d++;else if(s[i]!='\n'&&!(s[i]>='a'&&s[i]<='z')&&!(s[i]>='A'&&s[i]<='Z'))sc++;}printf("Spaces = %d\nDigits = %d\nSpecial characters = %d\n",sp,d,sc);
    return 0;
}
