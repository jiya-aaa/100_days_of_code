Q77. Check if the elements on the diagonal of a matrix are distinct.

#include <stdio.h>
int main(void) {
    int n,a[20][20],seen[401]={0},ok=1;scanf("%d",&n);for(int i=0;i<n;i++)for(int j=0;j<n;j++)scanf("%d",&a[i][j]);for(int i=0;i<n;i++){int x=a[i][i]+200;if(x<0||x>400||seen[x])ok=0;else seen[x]=1;}puts(ok?"Distinct":"Not distinct");
    return 0;
}
