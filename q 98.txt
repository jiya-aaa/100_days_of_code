#include <stdio.h>
#include <ctype.h>
#include <string.h>

int main() {
    char name[100];
    int i, lastSpace = -1;

    printf("Enter your name: ");
    fgets(name, sizeof(name), stdin);

    name[strcspn(name, "\n")] = '\0';

    for (i = 0; name[i] != '\0'; i++) {
        if (name[i] == ' ')
            lastSpace = i;
    }

    printf("Output: ");

    for (i = 0; i < lastSpace; i++) {
        if (i == 0 || name[i - 1] == ' ')
            printf("%c. ", toupper(name[i]));
    }

    for (i = lastSpace + 1; name[i] != '\0'; i++)
        printf("%c", name[i]);

    printf("\n");

    return 0;
}
