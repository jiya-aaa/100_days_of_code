Q9. Write a program to calculate simple and compound interest for given principal, rate, and time.

#include <stdio.h>
#include <math.h>
int main(void) {
    double p, r, t;
    scanf("%lf %lf %lf", &p, &r, &t);
    printf("Simple Interest = %.2f\n", p*r*t/100);
    printf("Compound Interest = %.2f\n", p*pow(1+r/100, t)-p);
    return 0;
}
