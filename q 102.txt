#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;

    vector<int> arr(n);
    for (int &value : arr) {
        cin >> value;
    }

    int x;
    cin >> x;

    auto it = lower_bound(arr.begin(), arr.end(), x);

    if (it == arr.end()) {
        cout << -1;
    } else {
        cout << (it - arr.begin());
    }

    return 0;
}
