Q73. Find the sum of each row of a matrix and store it in an array.

#include <stdio.h>
int main(void) {
    int r,c,a[20][20],sum[20]={0};scanf("%d%d",&r,&c);for(int i=0;i<r;i++)for(int j=0;j<c;j++){scanf("%d",&a[i][j]);sum[i]+=a[i][j];}for(int i=0;i<r;i++)printf("%d ",sum[i]);puts("");
    return 0;
}
